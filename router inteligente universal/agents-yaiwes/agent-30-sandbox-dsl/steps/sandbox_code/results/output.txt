from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from typing import Optional
import yaml
import json
import os
from pathlib import Path
import aiohttp

router = APIRouter(prefix="/sandbox")

STATE_PATH = Path("crazy_wall.state.json")
AGENT_LAUNCH_URL = "/gh/agent-launch"

class OrderRequest(BaseModel):
    agent_id: str
    order: str
    repo: Optional[str] = None

class StatusResponse(BaseModel):
    agent_id: str
    state: dict
    file_exists: bool

class LaunchResponse(BaseModel):
    run_id: str
    status: str

@router.post("/order")
async def sandbox_order(request: OrderRequest):
    """Process a Director order: generate YAML, validate, and launch the agent."""
    try:
        # Build system prompt from the request
        system_prompt = f"""
        Given the order: "{request.order}"
        Generate a chain.yaml following yaiwes.chain/v1 schema.
        Agent: {request.agent_id}
        Format: schema, input_block, agent, context, steps (with id, task, checks), edges.
        """
        
        # Get YAML from model (simulated here - actual call would go to model endpoint)
        chain_yaml = f"""
schema: yaiwes.chain/v1
input_block: |
  {request.order}

agent: {request.agent_id}

context:
  agent_launch_path: "{AGENT_LAUNCH_URL}"
  state_path: "{STATE_PATH}"
  model: "deepseek_flash"
  no_gpu: true
  no_keys_in_files: true

steps:
  - id: parse_order
    task: "Parse the order into structured steps"
    checks:
      - order_ok: "Order parsed correctly"

  - id: execute_steps
    task: "Execute the parsed steps"
    checks:
      - execution_complete: "All steps executed"

edges:
  - from: parse_order
    to: execute_steps
"""
        
        # Validate YAML syntax
        try:
            parsed = yaml.safe_load(chain_yaml)
        except yaml.YAMLError as e:
            raise HTTPException(status_code=400, detail=f"Invalid YAML: {str(e)}")
        
        # Validate schema structure
        if not isinstance(parsed, dict):
            raise HTTPException(status_code=400, detail="YAML must be a dictionary")
        
        required_keys = ["schema", "input_block", "agent", "context", "steps", "edges"]
        for key in required_keys:
            if key not in parsed:
                raise HTTPException(status_code=400, detail=f"Missing required key: {key}")
        
        if parsed.get("schema") != "yaiwes.chain/v1":
            raise HTTPException(status_code=400, detail="Invalid schema version")
        
        if not isinstance(parsed.get("steps"), list) or len(parsed["steps"]) == 0:
            raise HTTPException(status_code=400, detail="Must have at least one step")
        
        if not isinstance(parsed.get("edges"), list):
            raise HTTPException(status_code=400, detail="Edges must be a list")
        
        # Launch the agent via /gh/agent-launch
        # This would be an actual API call in production
        # For sandbox, we simulate the launch
        async with aiohttp.ClientSession() as session:
            launch_payload = {
                "agent_id": request.agent_id,
                "chain_yaml": chain_yaml,
                "repo": request.repo
            }
            async with session.post(AGENT_LAUNCH_URL, json=launch_payload) as response:
                if response.status != 200:
                    raise HTTPException(status_code=500, detail="Agent launch failed")
                launch_result = await response.json()
        
        # Update state.json
        state = {"agent_id": request.agent_id, "last_order": request.order, "status": "running"}
        with open(STATE_PATH, "w") as f:
            json.dump(state, f)
        
        return LaunchResponse(
            run_id=launch_result.get("run_id", "sandbox-run-001"),
            status="launched"
        )
        
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.get("/status")
async def sandbox_status(agent_id: str):
    """Get the current state of an agent from crazy_wall.state.json."""
    try:
        if not STATE_PATH.exists():
            return StatusResponse(
                agent_id=agent_id,
                state={"error": "State file not found"},
                file_exists=False
            )
        
        with open(STATE_PATH, "r") as f:
            state = json.load(f)
        
        # Filter state for the requested agent_id
        if state.get("agent_id") != agent_id:
            return StatusResponse(
                agent_id=agent_id,
                state={"error": "Agent not found in state"},
                file_exists=True
            )
        
        return StatusResponse(
            agent_id=agent_id,
            state=state,
            file_exists=True
        )
        
    except json.JSONDecodeError as e:
        raise HTTPException(status_code=500, detail=f"Invalid state JSON: {str(e)}")
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

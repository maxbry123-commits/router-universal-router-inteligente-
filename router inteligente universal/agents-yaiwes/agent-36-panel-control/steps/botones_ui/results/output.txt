  function actualizarReloj() {
    const ahora = new Date();
    relojPanel.textContent = ahora.toLocaleString('es-ES', {

      hour: '2-digit', minute: '2-digit', second: '2-digit'

  // --- Cargar lista de agentes desde el grupo orquestador (todas las API) ---
  async function cargarAgentes() {
    try {
      const resp = await fetch('/gh/groups/');
      if (!resp.ok) throw new Error('HTTP ' + resp.status);
      gruposCache = await resp.json();
      
      // Reconstruir map agente -> api_origen y llenar selects
      const agentesSet = new Set();
      apiOrigenMap = {};
      
      Object.entries(gruposCache).forEach(([grupo, cfg]) => {
        if (cfg.agentes && Array.isArray(cfg.agentes)) {
          cfg.agentes.forEach(ag => {
            agentesSet.add(ag);
            apiOrigenMap[ag] = cfg.api_origen || 'desconocido';
          });
        }
      });

      // Llenar selGrupo
      selGrupo.innerHTML = '<option value="">-- grupo --</option>';
      Object.keys(gruposCache).forEach(g => {
        const opt = document.createElement('option');
        opt.value = g;
        opt.textContent = g + ' (' + (gruposCache[g].api_origen || '?') + ')';
        selGrupo.appendChild(opt);
      });

      // Llenar selAgenteOrden con todos los agentes conocidos
      selAgenteOrden.innerHTML = '<option value="">-- agente --</option>';
      agentesSet.forEach(ag => {
        const opt = document.createElement('option');
        opt.value = ag;
        opt.textContent = ag + ' [' + (apiOrigenMap[ag] || '?') + ']';
        selAgenteOrden.appendChild(opt);
      });

      // Reset router si cambian grupos

      selAgenteRouter.disabled = true;

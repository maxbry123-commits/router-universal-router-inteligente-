<script>
  (function() {
    // --- estado y helpers ---
    let BASE = '';
    const LS = localStorage;
    const errorLine = document.getElementById('errorLine');

    function setError(msg) { errorLine.textContent = msg || ''; }
    function getKeys() { return { hf: LS.getItem('HF_TOKEN') || '', api: LS.getItem('X_API_KEY') || '' }; }
    function saveKeys(hf, api) { LS.setItem('HF_TOKEN', hf); LS.setItem('X_API_KEY', api); }

    // lectura de LIVE_URL desde flag
    async function fetchBaseUrl() {
      try {
        const resp = await fetch('https://raw.githubusercontent.com/maxbry123-commits/router-universal-router-inteligente-/main/router%20inteligente%20universal/agents-yaiwes/ROUTER_JOB_PAUSE.flag');
        const text = await resp.text();
        const match = text.match(/LIVE_URL=(.+)/);
        if (match) {
          let url = match[1].trim();
          if (url.endsWith('/')) url = url.slice(0, -1);
          BASE = url;
        } else {
          BASE = ''; 
          setError('no se pudo extraer LIVE_URL del flag');
        }
      } catch (e) {
        setError('error obteniendo LIVE_URL: ' + e.message);
        BASE = '';
      }
    }

    // peticiones con clave, headers
    async function apiFetch(path, options = {}) {
      const { hf, api } = getKeys();
      const headers = { 'Content-Type': 'application/json', ...options.headers };
      if (hf) headers['Authorization'] = `Bearer ${hf}`;
      if (api) headers['X-API-Key'] = api;
      const url = BASE + path;
      try {
        const res = await fetch(url, { ...options, headers });
        if (!res.ok) {
          let errBody = '';
          try { errBody = await res.text(); } catch(_) {}
          throw new Error(`status ${res.status}: ${errBody.slice(0,70)}`);
        }
        return res;
      } catch (err) {
        setError('API error: ' + err.message);
        throw err;
      }
    }

    // --- inicializar selects ---
    async function loadModels() {
      try {
        const res = await apiFetch('/v1/models');
        const data = await res.json();
        const sel = document.getElementById('modelSelect');
        sel.innerHTML = '';
        if (Array.isArray(data)) {
          data.forEach(m => {
            const opt = document.createElement('option');
            opt.value = m.id || m;
            opt.text = m.id || m;
            sel.add(opt);
          });
        } else if (data.data && Array.isArray(data.data)) {
          data.data.forEach(m => {
            const opt = document.createElement('option');
            opt.value = m.id;
            opt.text = m.id;
            sel.add(opt);
          });
        } else {
          sel.innerHTML = '<option>modelos no listados</option>';
        }
      } catch (e) {
        document.getElementById('modelSelect').innerHTML = '<option>error modelos</option>';
      }
    }

    async function loadAgents() {
      try {
        const res = await apiFetch('/chat/agents');
        const data = await res.json();
        const sel = document.getElementById('agentSelect');
        sel.innerHTML = '<option value="">sin agente</option>';
        if (Array.isArray(data)) {
          data.forEach(a => {
            const opt = document.createElement('option');
            opt.value = a.id || a.name || a;
            opt.text = a.name || a.id || a;
            sel.add(opt);
          });
        } else if (data.agents) {
          data.agents.forEach(a => {
            const opt = document.createElement('option');
            opt.value = a.id || a;
            opt.text = a.name || a.id || a;
            sel.add(opt);
          });
        }
      } catch (e) {
        document.getElementById('agentSelect').innerHTML = '<option>error agentes</option>';
      }
    }

    async function loadGhAccounts() {
      try {
        const res = await apiFetch('/gh/accounts');
        const data = await res.json();
        const sel = document.getElementById('ghSelect');
        sel.innerHTML = '';
        if (Array.isArray(data)) {
          data.forEach(g => {
            const opt = document.createElement('option');
            opt.value = g.login || g.id || g;
            opt.text = g.login || g.id || g;
            sel.add(opt);
          });
        } else if (data.accounts) {
          data.accounts.forEach(g => {
            const opt = document.createElement('option');
            opt.value = g.login || g;
            opt.text = g.login || g;
            sel.add(opt);
          });
        } else {
          sel.innerHTML = '<option>gh cuentas</option>';
        }
      } catch (e) {
        document.getElementById('ghSelect').innerHTML = '<option>error gh</option>';
      }
    }

    async function loadGroups() {
      try {
        const res = await apiFetch('/groups');
        const data = await res.json();
        const sel = document.getElementById('groupSelect');
        sel.innerHTML = '';
        if (Array.isArray(data)) {
          data.forEach(g => {
            const opt = document.createElement('option');
            opt.value = g.id || g.name || g;
            opt.text = g.name || g.id || g;
            sel.add(opt);
          });
        } else if (data.groups) {
          data.groups.forEach(g => {
            const opt = document.createElement('option');
            opt.value = g.id || g;
            opt.text = g.name || g.id || g;
            sel.add(opt);
          });
        } else {
          sel.innerHTML = '<option>grupos</option>';
        }
      } catch (e) {
        document.getElementById('groupSelect').innerHTML = '<option>error grupos</option>';
      }
    }

    // --- envio de mensajes al chat ---
    async function sendMessage() {
      const input = document.getElementById('chatInput');
      const text = input.value.trim();
      if (!text) return;
      const history = document.getElementById('history');
      const userDiv = document.createElement('div');
      userDiv.className = 'msg user';
      userDiv.textContent = text;
      history.appendChild(userDiv);
      history.scrollTop = history.scrollHeight;
      input.value = '';

      const model = document.getElementById('modelSelect').value;
      const agentOn = document.getElementById('agentSwitch').checked;
      const agentId = document.getElementById('agentSelect').value;
      let modelField = model;
      if (agentOn) modelField = model + '+agentes';

      const body = {
        model: modelField,
        messages: [{ role: 'user', content: text }],
        agent: agentId || undefined,
      };
      try {
        const res = await apiFetch('/v1/chat/completions', { method: 'POST', body: JSON.stringify(body) });
        const data = await res.json();
        let reply = '';
        if (data.choices && data.choices[0] && data.choices[0].message) reply = data.choices[0].message.content;
        else if (data.response) reply = data.response;
        else reply = JSON.stringify(data).slice(0,300);
        const asstDiv = document.createElement('div');
        asstDiv.className = 'msg assistant';
        asstDiv.textContent = reply;
        history.appendChild(asstDiv);
        history.scrollTop = history.scrollHeight;
      } catch (e) {
        const errDiv = document.createElement('div');
        errDiv.className = 'msg assistant';
        errDiv.textContent = '❌ error: ' + e.message;
        history.appendChild(errDiv);
      }
    }

    // --- subir archivo ---
    document.getElementById('attachBtn').addEventListener('click', () => {
      document.getElementById('fileInput').click();
    });
    document.getElementById('fileInput').addEventListener('change', async (e) => {
      const file = e.target.files[0];
      if (!file) return;
      const form = new FormData();
      form.append('file', file);
      form.append('chat_id', 'default');
      const { hf, api } = getKeys();
      const headers = {};
      if (hf) headers['Authorization'] = `Bearer ${hf}`;
      if (api) headers['X-API-Key'] = api;
      try {
        const res = await fetch(BASE + '/chat/documents', { method: 'POST', headers, body: form });
        if (!res.ok) throw new Error('upload fail ' + res.status);
        setError('archivo subido');
      } catch (err) {
        setError('error upload: ' + err.message);
      }
      e.target.value = '';
    });

    // --- workflow ➕ ---
    const wfPanel = document.getElementById('wflowPanel');
    document.getElementById('workflowPlusBtn').addEventListener('click', () => {
      wfPanel.style.display = (wfPanel.style.display === 'none') ? 'flex' : 'none';
    });
    document.getElementById('wfCancel').addEventListener('click', () => { wfPanel.style.display = 'none'; });
    document.getElementById('wfSave').addEventListener('click', async () => {
      const name = document.getElementById('wfName').value.trim();
      const repo = document.getElementById('wfRepo').value.trim();
      const plan = document.getElementById('wfPlan').value.trim();
      if (!name || !repo || !plan) { setError('completa nombre, repo, plan'); return; }
      try {
        const res = await apiFetch('/chat/storage', {
          method: 'POST',
          body: JSON.stringify({ name, repo, plan_path: plan })
        });
        if (res.ok) setError('workflow guardado');
        else setError('error al guardar workflow');
        wfPanel.style.display = 'none';
        document.getElementById('wfName').value = '';
        document.getElementById('wfRepo').value = '';
        document.getElementById('wfPlan').value = '';
      } catch (e) { setError(e.message); }
    });

    // --- panel control ---
    async function controlAction(path, label) {
      try {
        const res = await apiFetch(path, { method: 'POST' });
        if (res.ok) setError(label + ' OK');
        else setError(label + ' falló');
      } catch (e) { setError(e.message); }
    }
    document.getElementById('pauseBtn').addEventListener('click', () => controlAction('/control/pause-agents', 'pausar'));
    document.getElementById('resumeBtn').addEventListener('click', () => controlAction('/control/resume-agents', 'reanudar'));
    document.getElementById('emergencyBtn').addEventListener('click', () => controlAction('/control/emergency-stop', 'emergencia'));
    document.getElementById('startJobBtn').addEventListener('click', () => controlAction('/control/start-job', 'start job'));
    document.getElementById('orderBtn').addEventListener('click', () => controlAction('/control/order', 'orden programada'));

    // --- settings: guardar claves ---
    document.getElementById('saveKeysBtn').addEventListener('click', () => {
      const hf = document.getElementById('hfTokenInput').value.trim();
      const api = document.getElementById('apiKeyInput').value.trim();
      saveKeys(hf, api);
      setError('claves guardadas en localStorage');
      // recargar datos?
    });

    // cargar claves previas en inputs
    function loadKeysIntoInputs() {
      const { hf, api } = getKeys();
      document.getElementById('hfTokenInput').value = hf;
      document.getElementById('apiKeyInput').value = api;
    }

    // --- enviar mensaje con enter ---
    document.getElementById('sendBtn').addEventListener('click', sendMessage);
    document.getElementById('chatInput').addEventListener('keydown', (e) => { if (e.key === 'Enter') sendMessage(); });

    // --- init ---
    async function init() {
      await fetchBaseUrl();
      if (!BASE) { setError('LIVE_URL no definida'); }
      // carga inicial
      loadKeysIntoInputs();
      // No bloquear carga de selects si BASE no existe
      if (BASE) {
        loadModels();
        loadAgents();
        loadGhAccounts();
        loadGroups();
      } else {
        document.getElementById('modelSelect').innerHTML = '<option>sin url</option>';
      }
    }
    init();
  })();
</script>

  <input type="text" id="apiBase" placeholder="Ej: https://<job_id>--8000.hf.jobs">

  <label for="apiKey">API key del Router</label>
  <input type="password" id="apiKey" placeholder="Introduce la clave (solo en memoria)">

  <label for="msgInput">Mensaje</label>
  <textarea id="msgInput" rows="3" placeholder="Escribe tu mensaje aquí"></textarea>

  <button id="sendBtn">Enviar</button>
  <button id="hfLoginBtn">Iniciar sesión con Hugging Face</button>

  <div id="messages"></div>

<script>
  const apiBaseInput = document.getElementById('apiBase');
  const apiKeyInput = document.getElementById('apiKey');
  const msgInput = document.getElementById('msgInput');
  const sendBtn = document.getElementById('sendBtn');
  const hfLoginBtn = document.getElementById('hfLoginBtn');
  const messagesDiv = document.getElementById('messages');

  function addText(text, isError = false) {
    const p = document.createElement('p');
    p.className = isError ? 'error' : 'reply';
    p.textContent = text;
    messagesDiv.appendChild(p);
    messagesDiv.scrollTop = messagesDiv.scrollHeight;
  }

  sendBtn.addEventListener('click', async () => {
    const apiBase = apiBaseInput.value.trim();
    const apiKey = apiKeyInput.value.trim();
    const message = msgInput.value.trim();

    if (!apiBase) { addText('Debes indicar la URL base del Router.', true); return; }
    if (!apiKey) { addText('Debes introducir la API key del Router.', true); return; }
    if (!message) { addText('El mensaje no puede estar vacío.', true); return; }

    const url = apiBase.endsWith('/') ? apiBase + 'chat/send' : apiBase + '/chat/send';
    const payload = {
      message,
      provider: 'nvidia',
      model: 'nvidia/nemotron-3-super-120b-a12b',
      max_tokens: 1024
    };

    try {
      const response = await fetch(url, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-API-Key': apiKey
        },
        body: JSON.stringify(payload)
      });

      if (!response.ok) {
        const errText = await response.text();
        addText(`Error ${response.status}: ${errText}`, true);
        return;
      }

      const data = await response.json();
      const reply = data.reply || '(sin respuesta)';
      addText(`Respuesta: ${reply}`);
    } catch (err) {
      addText(`Error de red: ${err.message}`, true);
    }
  });

  hfLoginBtn.addEventListener('click', () => {
    // Marker for OAuth integration; actual token handling would be done via HF OAuth flow.
    window.location.href = 'https://huggingface.co/oauth/authorize';
  });
</script>

function initJobsPanel(container, fetchJson) {
  // Create textarea for JSON input
  const textarea = document.createElement('textarea');
  textarea.rows = 10;
  textarea.cols = 60;
  textarea.placeholder = 'Paste JSON array of jobs here';
  container.appendChild(textarea);

  // Create button
  const button = document.createElement('button');
  button.textContent = 'Ejecutar en paralelo';
  container.appendChild(button);

  // Create output container for results
  const output = document.createElement('div');
  container.appendChild(output);

  // Handle click
  button.addEventListener('click', function() {
    let jobs;
    try {
      jobs = JSON.parse(textarea.value);
      if (!Array.isArray(jobs)) throw new Error('Expected a JSON array');
    } catch (e) {
      output.textContent = 'Error: ' + e.message;
      return;
    }

    fetchJson('/chat/jobs/run', {
      method: 'POST',
      body: JSON.stringify({jobs: jobs, max_parallel: 8})
    }).then(function(result) {
      if (result.status >= 200 && result.status < 300) {
        const data = result.body;
        const lines = data.map(function(job) {
          return job.id + ' - ' + job.status;
        });
        output.textContent = lines.join('\n');
      } else {
        output.textContent = 'Error: HTTP ' + result.status;
      }
    }).catch(function(err) {
      output.textContent = 'Error: ' + err.message;
    });
  });
}

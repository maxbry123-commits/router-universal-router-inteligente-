## Cableado propuesto

```python
# router_yaiwes.py - Router YAIWES (apoyo a Agent 17, no pisa archivos activos)

from integration.chat_mvp import app as chat_app
from integration.huggingface import provider as hf_provider
from integration.chat_mvp.models import models_catalog
from integration.chat_mvp.completions import chat_completions
from integration.chat_mvp.selector import model_selector
from integration.chat_mvp.routing import route_to_provider, fallback_handler

# --- GET /v1/models ---
# Reutiliza catálogo existente
@chat_app.get("/v1/models")
async def list_models():
    return models_catalog.get_all()

# --- POST /v1/chat/completions ---
# Reutiliza endpoint existente con routing
@chat_app.post("/v1/chat/completions")
async def create_chat_completion(request):
    # Selector de modelo/proveedor
    selected = model_selector.select(request.model)
    # Routing al proveedor (con agente/sin agente)
    if request.get("agent_mode", False):
        result = await route_to_provider(selected, request, agent=True)
    else:
        result = await route_to_provider(selected, request, agent=False)
    return result

# Fallback: solo cuando está autorizado por el router
@chat_app.exception_handler(Exception)
async def global_fallback(request, exc):
    if fallback_handler.is_authorized(exc):
        return await fallback_handler.process(exc)
    # Fail-closed: error sin secretos
    return {"error": "internal_error", "message": "Request failed"}, 500
```

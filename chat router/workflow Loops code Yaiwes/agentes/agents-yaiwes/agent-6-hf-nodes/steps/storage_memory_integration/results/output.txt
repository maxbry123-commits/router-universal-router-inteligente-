CABLEADO:
```
Open WebUI <-> /chat endpoints (router.py) <-> Store (store.py)
                |
                +-> SQLite (conversaciones, mensajes, documentos, grafo, cache)
                +-> PostgreSQL (verdad durable: usuarios, auditorías, referencias)
                +-> Redis (estado efímero: caché TTL, sesiones, locks)
                +-> Graphiti/FalkorDB (memoria relacional temporal)
                +-> AgentDB (memoria agentic: episodios, patrones)
                +-> HF Bucket (archivos originales, adjuntos, snapshots)
                +-> Secret Bank (vault_api.py - referencias, nunca secretos planos)
```

PRUEBAS:
```python
def test_persistence(store, db_postgres, redis_client, graphiti, agentdb, bucket):
    # 1. Crear conversación
    cid = store.new_conversation("Test", "orquestador-g0", "test_user")
    assert cid is not None
    
    # 2. Guardar mensajes
    store.add_message(cid, "user", "Hola mundo")
    store.add_message(cid, "assistant", "Respuesta de prueba", "hf", "gpt2")
    
    # 3. Adjuntar archivo
    doc = store.put_document("test.txt", "text/plain", b"contenido de prueba", cid)
    assert doc["id"] is not None
    
    # 4. Guardar memoria (PostgreSQL)
    db_postgres.save_memory_entry(cid, "test_user", {"type": "chat", "summary": "Prueba exitosa"})
    
    # 5. Guardar memoria (Redis)
    redis_client.setex(f"session:test_user", 3600, cid)
    
    # 6. Guardar memoria (Graphiti/FalkorDB)
    graphiti.add_entity("test_user", "conversation", cid)
    graphiti.link("test_user", cid, "had")
    
    # 7. Guardar memoria (AgentDB)
    agentdb.record_episode(cid, "test_agent", "user_message", "assistant_response")
    
    # 8. Cerrar chat
    del cid
    
    # 9. Reabrir y recuperar conversación
    recovered = store.conversation(doc["id"])
    assert recovered is not None
    msgs = store.messages(recovered["id"])
    assert len(msgs) == 2
    assert msgs[1]["content"] == "Respuesta de prueba"
    
    # 10. Recuperar archivo
    recovered_doc = store.document(doc["id"])
    assert recovered_doc["name"] == "test.txt"
    assert store.document_text(doc["id"]) == "contenido de prueba"
    
    # 11. Recuperar memoria/contexto
    from_postgres = db_postgres.get_memory_entries(cid)
    assert len(from_postgres) >= 1
```

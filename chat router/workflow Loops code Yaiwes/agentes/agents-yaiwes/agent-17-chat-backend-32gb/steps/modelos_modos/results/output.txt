import os
import asyncio
import aiohttp
import json
from typing import Optional, Dict, List, Any
from fastapi import FastAPI, HTTPException, Request
from pydantic import BaseModel
import store

app = FastAPI()

HF_TOKEN = os.environ.get("HF_TOKEN", "")
RIU_ROUTER_API_KEY = os.environ.get("RIU_ROUTER_API_KEY", "")
ROUTER_BASE_URL = os.environ.get("ROUTER_BASE_URL", "http://localhost:8000")

class ChatRequest(BaseModel):
    model: str
    messages: List[Dict[str, str]]
    stream: Optional[bool] = False

@app.get("/v1/models")
async def get_models():
    return {
        "object": "list",
        "data": [
            {
                "id": "deepseek-flash",
                "object": "model",
                "created": 1234567890,
                "owned_by": "ria",
                "permission": []
            },
            {
                "id": "deepseek-flash+agentes",
                "object": "model",
                "created": 1234567890,
                "owned_by": "ria",
                "permission": []
            }
        ]
    }

@app.post("/v1/chat/completions")
async def chat_completion(request: ChatRequest, fastapi_request: Request):
    auth_header = fastapi_request.headers.get("Authorization", "")
    api_key = fastapi_request.headers.get("X-API-Key", "")
    
    if not auth_header.startswith("Bearer ") or not api_key:
        raise HTTPException(status_code=401, detail="Invalid authentication")
    
    token = auth_header.replace("Bearer ", "")
    if token != HF_TOKEN or api_key != RIU_ROUTER_API_KEY:
        raise HTTPException(status_code=401, detail="Invalid credentials")
    
    model = request.model
    messages = request.messages
    last_message = messages[-1]["content"] if messages else ""
    
    # Save user message
    store.save_message("user", "user", last_message)
    
    if model.endswith("+agentes"):
        router_endpoint = "/chat/agents"
    else:
        router_endpoint = "/chat/send"
    
    headers = {
        "Authorization": f"Bearer {HF_TOKEN}",
        "X-API-Key": RIU_ROUTER_API_KEY,
        "Content-Type": "application/json"
    }
    
    async with aiohttp.ClientSession() as session:
        async with session.post(
            f"{ROUTER_BASE_URL}{router_endpoint}",
            json={"messages": messages},
            headers=headers
        ) as resp:
            if resp.status != 200:
                error_text = await resp.text()
                raise HTTPException(status_code=resp.status, detail=error_text)
            
            response_data = await resp.json()
    
    # Save assistant message
    assistant_text = response_data.get("response", response_data.get("text", ""))
    store.save_message("assistant", model, assistant_text)
    
    return {
        "id": "chatcmpl-" + os.urandom(12).hex(),
        "object": "chat.completion",
        "created": int(asyncio.get_event_loop().time()),
        "model": model,
        "choices": [
            {
                "index": 0,
                "message": {
                    "role": "assistant",
                    "content": assistant_text
                },
                "finish_reason": "stop"
            }
        ],
        "usage": {
            "prompt_tokens": 0,
            "completion_tokens": 0,
            "total_tokens": 0
        }
    }
